# Ambulance CAD Frontend

Production-style React/Vite frontend scaffold for an ambulance/EMS Computer-Aided Dispatch system.

## Run

```bash
npm install
npm run dev
```

## Demo authentication

The login screen writes `cad_role` to `localStorage`. Supported keys:

- `call-taker`
- `dispatcher`
- `paramedic`
- `hospital`
- `operations`
- `fleet`
- `admin`

## Architecture

- `src/context/CadContext.jsx` is the mock domain state boundary.
- Replace its mock actions with API clients/WebSocket subscriptions when the backend is ready.
- `MapPanel.jsx` is the Leaflet integration boundary. It currently renders a deterministic CAD map surface and accepts `units`, `incidents`, `routes`, and `markers`-style data through props.
- OSRM should provide route/ETA data; MQTT/WebSocket should provide live unit state; FastAPI should expose REST/domain commands; PostGIS should remain the geospatial persistence layer.
- No routing algorithm or OSRM behavior is faked.

## Role routes

`/login`, `/call-taker`, `/dispatcher`, `/paramedic`, `/hospital`, `/operations`, `/fleet`, `/admin`.

Route guards enforce the selected demo role and redirect unauthorized workspace access back to the user's own role.

## Visual system

Dark navy CAD palette, 8px spacing rhythm, 60px global header, thin borders, high-contrast emergency priority/status states, desktop command-center layouts, and mobile-oriented Paramedic workspace.

import { createContext, useContext, useMemo, useState } from 'react'
import { initialIncidents } from '../data/mockIncidents'
import { initialUnits } from '../data/mockUnits'
import { hospitals } from '../data/mockHospitals'
import { users } from '../data/mockUsers'
import { maintenance } from '../data/mockMaintenance'
import { initialMessages } from '../data/mockMessages'

const CadContext = createContext(null)
export function CadProvider({ children }) {
  const [incidents, setIncidents] = useState(initialIncidents)
  const [units, setUnits] = useState(initialUnits)
  const [messages, setMessages] = useState(initialMessages)
  const [notifications, setNotifications] = useState([])
  const [currentUser, setCurrentUser] = useState(() => {
    const role = localStorage.getItem('cad_role')
    return users.find(u => u.roleKey === role) || users[0]
  })

  const loginAs = roleKey => {
    const user = users.find(u => u.roleKey === roleKey) || users[0]
    localStorage.setItem('cad_role', user.roleKey)
    setCurrentUser(user)
  }
  const logout = () => { localStorage.removeItem('cad_role'); setCurrentUser(null) }

  const addIncident = payload => {
    const nextNumber = Math.max(...incidents.map(i => Number(i.id.replace('CAD-',''))), 2988) + 1
    const incident = { ...payload, id:`CAD-${nextNumber}`, time:new Date().toLocaleTimeString([], {hour:'2-digit',minute:'2-digit',second:'2-digit'}), status:'Pending', notes:[] }
    setIncidents(prev => [incident, ...prev])
    setNotifications(prev => [{ id:Date.now(), type:'success', message:`${incident.id} submitted to dispatch queue.` }, ...prev].slice(0,4))
    return incident
  }
  const appendNote = (incidentId, note) => {
    if (!note.trim()) return
    setIncidents(prev => prev.map(i => i.id === incidentId ? {...i, notes:[...(i.notes||[]), {text:note, at:new Date().toLocaleTimeString()}]} : i))
    setNotifications(prev => [{id:Date.now(),type:'success',message:`Note appended to ${incidentId}.`},...prev].slice(0,4))
  }
  const assignUnit = (incidentId, unitId) => {
    setIncidents(prev => prev.map(i => i.id === incidentId ? {...i, assignedUnit:unitId, status:'Dispatched'} : i))
    setUnits(prev => prev.map(u => u.id === unitId ? {...u,status:'EN ROUTE',assignedIncident:incidentId} : u))
    setNotifications(prev => [{id:Date.now(),type:'success',message:`${unitId} dispatched to ${incidentId}.`},...prev].slice(0,4))
  }
  const updateUnitStatus = (unitId,status) => setUnits(prev => prev.map(u => u.id===unitId?{...u,status}:u))
  const sendMessage = (message, to='ALL') => {
    if (!message.trim()) return
    setMessages(prev => [{id:Date.now(),from:currentUser?.name||'CAD',to,text:message,time:new Date().toLocaleTimeString()},...prev])
  }
  const dismissNotification = id => setNotifications(prev => prev.filter(n=>n.id!==id))

  const value = useMemo(() => ({ incidents,setIncidents,units,setUnits,hospitals,users,maintenance,messages,notifications,currentUser,loginAs,logout,addIncident,appendNote,assignUnit,updateUnitStatus,sendMessage,dismissNotification }), [incidents,units,messages,notifications,currentUser])
  return <CadContext.Provider value={value}>{children}</CadContext.Provider>
}
export const useCad = () => useContext(CadContext)
